  state "300ms warmup\nafter reset_track publish" as WARM
}

state OFFBOARD as "OFFBOARD\n(NAVIGATION_STATE_OFFBOARD)" #LightGreen

state "Stick Override" as STICK_OVERRIDE

[*] --> POSCTL : system boot

POSCTL --> WAIT_RC : RC valid + position valid
WAIT_RC --> WARMUP : evaluateTrigger() = true
note right of WAIT_RC
  publishResetTrack()
  pending_offboard_request_ = true
end note

WARMUP --> OFFBOARD : elapsed >= 300ms
note right of WARMUP
  VehicleCommand(DO_SET_MODE, param2=6)
end note

OFFBOARD --> STICK_OVERRIDE : Stick movement
STICK_OVERRIDE --> POSCTL : nav_state changes

OFFBOARD --> POSCTL : Heartbeat timeout

' Inference Lifecycle
state "InferenceNode Lifecycle" as LIFECYCLE {
  state "Unconfigured" as UNCONF #Pink
  state "Inactive" as INACT #LightYellow
  state "Active" as ACT #LightGreen

  [*] --> UNCONF
  UNCONF --> INACT : on_configure()
  INACT --> ACT : on_activate()
  ACT --> INACT : on_deactivate()
  INACT --> UNCONF : on_cleanup()

    state "Inference Loop @ 40 Hz" as LOOP {

      ADV --> OBS : advance_step()
      OBS --> ACTOR : get_all_features()
      ACTOR --> PROC : policy_actor(obs)
      PROC --> PUB : process_action()
      PUB --> [*]
    }
  }
}
